## Terminal 2 - 키보드 주행 및 pause toggle

python -u ai_worker_pedal_teleop.py \
  --keyboard \
  --domain-id 30 \
  --linear-speed 0.4 \
  --angular-speed 0.8 \
  --lift-speed 0.08 \
  --publish-rate 30

python ai_worker_pedal_lift_override.py --allow-lift-without-tracking --keyboard

# W/S: 전진/후진, A/D: 좌/우 회전, O/P: 리프트 상승/하강
# U: pause toggle
#   첫 번째 U: base/lift 정지 + 상체 자세 유지 + replay 일시정지
#   두 번째 U: pause 해제 후 추적 안정 시 teleop/replay 자동 재개
#   R은 최초 Vision Pro 활성화 때만 필요; 이후 추적 장애는 안정화 후 자동 복구

## Terminal 3 대안 - Vision Pro/Physical AI Tools 없이 leader master를 Loop로만 기록

# Terminal 1에서 ffw_sg2_ai를 실행한 경우에 사용.
# --loop, --arm AI_WORKER, --ee ai_worker_2g, --no-record는 자동 적용됨.
python -u teleop_hand_and_arm.py \
  --leader-master-loop \
  --loop-addr <loop-host>:50051 \
  --ai-worker-ros-domain-id 30 \
  --frequency 30

# 이 터미널에서는 R/S를 누르지 않음.
# Config Loop UI에서 episode별 task prompt를 지정하고 기록 start/stop.
# source_valid에 follower/leader/head/lift/mobile 수신 상태가 기록됨.
# 초기 동기화/차렷자세는 ffw_sg2_ai와 leader master가 담당하며,
# 이 프로세스는 로봇 command를 publish하지 않음.
# 종료 순서: Loop episode stop -> leader 정상 종료/자세 복귀 -> Ctrl+C.

## Episode replay

EPISODE="/home/kimm/Downloads/xr_tele/teleop/utils/data/pick cube/episode_0002"

# 1. 로봇을 움직이지 않고 Rerun 그래프/카메라 확인
python -u ai_worker_replay.py "$EPISODE" \
  --dry-run \
  --no-hand \
  --visualize \
  --rerun-viewer native \
  --viewer-hold-seconds 30

# 2. 실기 replay
# Terminal 2의 keyboard 명령을 먼저 실행해 둬야 함
# teleop_hand_and_arm.py는 종료해서 두 프로세스가 팔을 동시에 제어하지 않게 할 것
python -u ai_worker_replay.py "$EPISODE" \
  --execute \
  --domain-id 30 \
  --no-hand \
  --visualize \
  --rerun-viewer native \
  --post-replay-prompt \
  --ai-worker-model sg2 \
  --home-duration 3

# replay 중 U: pause/resume toggle
# replay 종료 후 R: 다시 replay
# replay 종료 후 Q: 3초 동안 차렷자세로 이동 후 종료
